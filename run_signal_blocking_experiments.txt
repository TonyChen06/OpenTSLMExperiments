################################################################################
# Signal Blocking Experiments (real-trained models, random window blocking)
#
# Purpose: Evaluate how models degrade when random windows of the signal are
# replaced with straight-line interpolation between boundary values. This
# erases diagnostic features (QRS morphology, ST changes, etc.) while keeping
# the signal visually continuous.
#
# Algorithm: For each signal, randomly place non-overlapping blocks whose
# lengths ~ Normal(avg, std). Within each block, replace the signal with
# linspace(signal[start], signal[end]) — a straight line.
#
# ECG-specific: 10s signal at 100Hz = 1000 samples
#   Blocking 3s = 30% of the signal erased
#   Blocking 5s = 50% of the signal erased
#
# Checkpoints (same as partial noise experiments):
#   TSQA:  results/Llama_3_2_1B/OpenTSLMFlamingo_real_tsqa/stage1_mcq/checkpoints/best_model.pt
#   HAR:   hf_checkpoints/har_model_checkpoint.pt
#   Sleep: results/Llama_3_2_1B/OpenTSLMFlamingo_real_sleep/stage4_sleep_cot/checkpoints/best_model.pt
#   ECG:   hf_checkpoints/model_checkpoint.pt
#
# First 20 samples' COT rationales saved to output JSON.
################################################################################


# ==============================================================================
# ECG — Block 3s out of 10s (30% blocked, avg 0.5s blocks)
# ==============================================================================
PYTHONPATH=src:src/open_flamingo CUDA_VISIBLE_DEVICES=0 python evaluation/opentslm/ecg_qa_cot/evaluate_ecg_flamingo.py \
    --checkpoint hf_checkpoints/model_checkpoint.pt \
    --use_block \
    --block_total_sec 3.0 \
    --block_avg_sec 0.5 \
    --block_std_sec 0.1 \
    --block_seed 67 \
    --output results/eval_ecg_block_3s.json
